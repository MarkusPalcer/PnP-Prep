\documentclass[12pt]{pnpprep}

\begin{document}

\begin{blueprint}{parent}
    \field{name}

    \registerChildEnvironment{child}{childenvironment}

    \begin{render}
        Parent: \namevalue

        \renderChildren
    \end{render}
\end{blueprint}


\begin{blueprint}{childenvironment}
    \field{name}

    \begin{render}
        Child: \namevalue \par
    \end{render}
\end{blueprint}


\begin{parent}
    \name{Parent 1}

    \begin{childGroup}
        \begin{child}
            \name{Child 1}
        \end{child}
        \begin{child}
            \name{Child 2}
        \end{child}
    \end{childGroup}

    \begin{childGroup}
        \begin{child}
            \name{Child 3}
        \end{child}
        \begin{child}
            \name{Child 4}
        \end{child}
    \end{childGroup}
\end{parent}

\begin{parent}

    \begin{childGroup}

        \end{child}

            \name{Child 2}
        \end{child}
    \end{childGroup}

        \end{child}
    \end{childGroup}
\end{parent}

\end{document}

\documentclass[12pt]{pnpprep}

\begin{document}

\begin{blueprint}{parent}
    \field{name}

    \registerChildEnvironment{child}{childenvironment}

    \begin{render}
        Parent: \namevalue

        \renderChildren
    \end{render}
\end{blueprint}


\begin{blueprint}{childenvironment}
    \field{name}

    \begin{render}
        Child: \namevalue
    \end{render}
\end{blueprint}


\begin{parent}
    \name{P1}

    \begin{child}
        \name{C1}
    \end{child}

\end{parent}


\begin{parent}
    \name{P2}

    \begin{child}
        \name{C2}
    \end{child}

\end{parent}

\end{document}

% Child: C1

\documentclass{article}
\usepackage[a6paper,landscape,margin=5mm,noheadfoot]{geometry}

\pagestyle{empty}
\setlength{\parindent}{0pt}
\setlength{\parskip}{0pt}

\newdimen\frontbodyheight
\newdimen\backbodyheight
\frontbodyheight=75mm
\backbodyheight=90mm

\newbox\cardcontent
\newbox\frontcontent
\newbox\backcontent

\newcommand{\CardExplicitBreak}{%
    \par
    \mark{EXPLICIT-CARD-BREAK}%
    \penalty-10000
}

\newcommand{\ProbeLine}[1]{%
    \rule{0pt}{8mm}\texttt{#1}\par
}

\newcommand{\PrintFront}[3]{%
    \thispagestyle{empty}%
    \noindent
    \fbox{%
        \begin{minipage}[c][\dimexpr15mm-2\fboxsep-2\fboxrule\relax][c]{%
            \dimexpr\textwidth-2\fboxsep-2\fboxrule\relax
        }
            \textbf{#1}\par
            #3
        \end{minipage}%
    }\par
    \vspace{2mm}%
    \noindent
    \fbox{%
        \begin{minipage}[t][\frontbodyheight][t]{%
            \dimexpr\textwidth-2\fboxsep-2\fboxrule\relax
        }
            \unvbox#2
        \end{minipage}%
    }\par
    \newpage
}

\newcommand{\PrintBack}[1]{%
    \thispagestyle{empty}%
    \noindent
    \fbox{%
        \begin{minipage}[t][\backbodyheight][t]{%
            \dimexpr\textwidth-2\fboxsep-2\fboxrule\relax
        }
            \unvbox#1
        \end{minipage}%
    }\par
    \newpage
}

\newcommand{\RunCard}[2]{%
    \begingroup
        \setbox\cardcontent=\vbox{%
            \hsize=\dimexpr\textwidth-2\fboxsep-2\fboxrule\relax
            \parindent=0pt
            \parskip=0pt
            \let\newpage\CardExplicitBreak
            #2\par
        }%
        \setbox\frontcontent=\vsplit\cardcontent to \frontbodyheight
        \edef\actualbreak{\splitbotmark}%
        \def\expectedbreak{EXPLICIT-CARD-BREAK}%
        \def\breakstatus{}%
        \ifx\actualbreak\expectedbreak
            \def\breakstatus{Explicit break marker detected}%
        \fi
        \PrintFront{#1}{\frontcontent}{\breakstatus}%
        \ifvoid\cardcontent
            \null
            \newpage
        \else
            \setbox\backcontent=\vsplit\cardcontent to \backbodyheight
            \PrintBack{\backcontent}%
            \ifvoid\cardcontent
            \else
                \PackageError{page-splitting-prototype}%
                    {Prototype content exceeds two pages}%
                    {Reduce the probe content for this feasibility test.}%
            \fi
        \fi
    \endgroup
}

\end{document}

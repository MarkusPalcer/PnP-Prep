\documentclass[12pt]{pnpprep}

\begin{document}

\begin{blueprint}{creature}
    \field{name}
    \field[optional]{courage}

    \registerChildEnvironment{weapon}{weaponBlueprint}
    \registerChildEnvironment{armor}{armorBlueprint}

    \begin{render}
        \begin{panel}{\namevalue}
            Name: \namevalue
            \par

            \ifpresent{courage}
                {Courage: \couragevalue}

            \ifmissing{courage}
                {Courage: unknown}

            \par

            \renderChildren
        \end{panel}
    \end{render}
\end{blueprint}


\begin{blueprint}{weaponBlueprint}
    \field{name}
    \field{damage}
    \field[optional]{modifier}

    \begin{render}
        Weapon: \namevalue
        \par
        Damage: \damagevalue
        \par

        \ifpresent{modifier}
            {Modifier: \modifiervalue}

        \ifmissing{modifier}
            {Modifier: none}

        \par
    \end{render}
\end{blueprint}


\begin{blueprint}{armorBlueprint}
    \field{name}
    \field{protection}

    \begin{render}
        Armor: \namevalue
        \par
        Protection: \protectionvalue
        \par
    \end{render}
\end{blueprint}


\begin{creature}
    \name{Goblin}
    \courage{8}

    \begin{weapon}
        \name{Club}
        \damage{1W+2}
    \end{weapon}

    \begin{armor}
        \name{Hide}
        \protection{1}
    \end{armor}
\end{creature}


\end{document}

            \ifpresent{PA}{\textbf{PA: }\PAvalue}\quad
            \textbf{TP: }\TPvalue\quad
            \textbf{INI: }\INIvalue\quad
            \par
            
            \ifpresent{ruestung}{
                \textbf{\ruestungvalue :}\quad
                \ifpresent{RS}{\textbf{RS: }\RSvalue}\quad
                \ifmissing{RS}{\textbf{RS: }0}\quad
                \par
            }
        \end{FlexibleBox}
    \end{render}
\end{databox}

\cs_new_protected:Npn \pnp_blueprint_load:nn #1 #2
{
  \sys_if_platform_windows:TF
  {
    \pnp_blueprint_load_windows:nn {#1}{#2}
  }
  {
    \pnp_blueprint_load_linux:nn {#1}{#2}
  }
}


\cs_new_protected:Npn \pnp_blueprint_load_windows:nn #1 #2
{
  \msg_fatal:nn
    { pnp-blueprint }
    { windows-not-supported }
}

\cs_new_protected:Npn \pnp_blueprint_load_linux:nn #1 #2
{
  % Get absolute current directory
  \sys_get_shell:nnN
    { pwd }
    { }
    \l_pnp_blueprint_load_current_dir_tl

  \tl_trim_spaces:V \l_pnp_blueprint_load_current_dir_tl

  \iow_term:x
  {
    Current~directory:~
    \l_pnp_blueprint_load_current_dir_tl
  }

  % Search blueprint file (use realpath via -exec to get an absolute path)
  \sys_get_shell:nnN
  {
    find~.~-type~f~\(~-name~#2.#1~-o~-path~*/#1/#2.tex~\)~-exec~realpath~{}~\;~-quit
  }
  { }
  \l_pnp_blueprint_load_found_file_tl

  \tl_trim_spaces:V \l_pnp_blueprint_load_found_file_tl

  % If nothing found locally, search PNP_LIBRARY (if set) and return absolute path
  \tl_if_blank:VTF \l_pnp_blueprint_load_found_file_tl
  {
    % First, try to find the packaged helper script via kpsewhich so the file
    % is resolved from the TeX tree (TEXMFHOME/TEXMFLOCAL) rather than a
    % hard-coded relative path. This avoids brittle relative paths and works
    % even when the package is installed.
    % build kpsewhich command with explicit space token to avoid concatenation
    \tl_set:Nx \l_pnp_blueprint_load_kpse_cmd_tl { kpsewhich \c_space_tl tex/latex/pnpprep/blueprint/findlib.sh }
    \exp_args:NV \sys_get_shell:nnN \l_pnp_blueprint_load_kpse_cmd_tl { } \l_pnp_blueprint_load_script_path_tl
    \tl_trim_spaces:V \l_pnp_blueprint_load_script_path_tl

    \tl_if_blank:VTF \l_pnp_blueprint_load_script_path_tl
    {
      % kpsewhich failed; fall back to local helper path in the workspace
      \iow_term:x { Find-helper:~using~local~helper }
      \sys_get_shell:nnN { ./tex/latex/pnpprep/blueprint/findlib.sh~#1~#2 } { } \l_pnp_blueprint_load_found_file_tl
    }
    {
      % kpsewhich returned an absolute path to the script. Construct a
      % command token-list and pass it to the shell using NV expansion so
      % spaces in the script path are preserved correctly.
      \iow_term:x { Find-helper:~kpsewhich~found~\l_pnp_blueprint_load_script_path_tl }
      \tl_set:Nx \l_pnp_blueprint_load_cmd_tl { \l_pnp_blueprint_load_script_path_tl \c_space_tl #1 \c_space_tl #2 }
      \exp_args:NV \sys_get_shell:nnN \l_pnp_blueprint_load_cmd_tl { } \l_pnp_blueprint_load_found_file_tl
    }

    \tl_trim_spaces:V \l_pnp_blueprint_load_found_file_tl
  }
  {
    % found locally, nothing to do
  }

  \tl_if_blank:VTF
    \l_pnp_blueprint_load_found_file_tl
  {
    \msg_warning:nnnn
      { pnp-blueprint }
      { blueprint-not-found }
      {#1}
      {#2}
  }
  {
    \msg_info:nnnnn
      { pnp-blueprint }
      { blueprint-loaded }
      {#1}
      {#2}
      {\l_pnp_blueprint_load_found_file_tl}

    \input{\l_pnp_blueprint_load_found_file_tl}
  }
}
